        function loadNextMathQuestion() {
            const questions = mathBank[currentLevel] || mathBank[1];

            document.getElementById('math-progress-badge').innerText = `SOAL ${mathSession.currentCorrect + 1} / ${mathSession.targetCorrect}`;
            document.getElementById('math-visual-box').innerHTML = qObj.visual;
            document.getElementById('math-question').innerText = qObj.q;

            const feedbackEl = document.getElementById('math-feedback');
            feedbackEl.classList.add('hidden');
            feedbackEl.innerHTML = '';

            const grid = document.getElementById('math-options-grid');
            grid.innerHTML = qObj.opts.map(opt => {
                const optVal = typeof opt === 'string' ? `'${opt.replace(/'/g, "\\'")}'` : opt;
                const ansVal = typeof qObj.ans === 'string' ? `'${qObj.ans.replace(/'/g, "\\'")}'` : qObj.ans;

        let isDrawing = false;

        function initScratchpad() {
            const padCanvas = document.getElementById('scratchpadCanvas');
            padCtx = padCanvas.getContext('2d');
            padCtx.strokeStyle = '#38bdf8';
            padCtx.lineWidth = 3;
            padCtx.lineCap = 'round';

            const getPos = (e) => {
                const rect = padCanvas.getBoundingClientRect();
                const clientX = e.touches ? e.touches[0].clientX : e.clientX;
                const clientY = e.touches ? e.touches[0].clientY : e.clientY;
                return {
                    x: (clientX - rect.left) * (padCanvas.width / rect.width),
                    y: (clientY - rect.top) * (padCanvas.height / rect.height)
                };
            };

            padCanvas.onmousedown = (e) => { isDrawing = true; const p = getPos(e); padCtx.beginPath(); padCtx.moveTo(p.x, p.y); };
            padCanvas.onmousemove = (e) => { if (isDrawing) { const p = getPos(e); padCtx.lineTo(p.x, p.y); padCtx.stroke(); } };
            padCanvas.onmouseup = () => { isDrawing = false; };

            padCanvas.ontouchstart = (e) => { isDrawing = true; const p = getPos(e); padCtx.beginPath(); padCtx.moveTo(p.x, p.y); };
            padCanvas.ontouchmove = (e) => { if (isDrawing) { const p = getPos(e); padCtx.lineTo(p.x, p.y); padCtx.stroke(); } };
            padCanvas.ontouchend = () => { isDrawing = false; };
        }

        function update() {
            if (gameState !== 'PLAYING') return;

            if (keys.left) {
                player.vx = -player.speed;
                player.facing = 'LEFT';
            } else if (keys.right) {
                player.vx = player.speed;
                player.facing = 'RIGHT';
            } else {
                player.vx *= 0.7;
            }

            player.vy += 0.55;
            player.x += player.vx;
            player.y += player.vy;

            if (player.invulnerableTimer > 0) player.invulnerableTimer--;

            player.grounded = false;
            platforms.forEach(p => {
                if (player.x + player.width > p.x &&
                    player.x < p.x + p.w &&
                    player.y + player.height > p.y &&
                    player.y + player.height < p.y + p.h + player.vy + 4 &&
                    player.vy >= 0) {
                    player.grounded = true;
                    player.vy = 0;
                    player.y = p.y - player.height;
                }
            });

            cameraX = player.x - 200;
            if (cameraX < 0) cameraX = 0;

            if (player.y > 540) {
                takeDamage();
                player.x = 50;
                player.y = 300;
                player.vy = 0;
            }

            coins.forEach(c => {
                if (!c.taken && Math.abs(player.x - c.x) < 25 && Math.abs(player.y - c.y) < 30) {
                    c.taken = true;
                    gold += 10;
                    playSfx('coin');
                    updateHUD();
                }
            });

            gems.forEach(g => {
                if (!g.taken && Math.abs(player.x - g.x) < 30 && Math.abs(player.y - g.y) < 30) {
                    g.taken = true;
                    gemsCollected++;
                    playSfx('coin');
                    updateHUD();
                }
            });

            mathChests.forEach(c => {
                if (!c.opened && Math.abs(player.x - c.x) < 35 && Math.abs(player.y - c.y) < 35) {
                    openMathLockModal(c);
                }
            });

            slimes.forEach(s => {
                if (!s.alive) return;
                s.x += s.vx;
                if (s.x > s.maxX || s.x < s.minX) s.vx *= -1;

                if (Math.abs(player.x - s.x) < 28 && Math.abs(player.y - s.y) < 28) {
                    if (player.vy > 0 && player.y + player.height < s.y + 15) {
                        s.alive = false;
                        player.vy = -8;
                        playSfx('coin');
                        gold += 15;
                        updateHUD();
                    } else {
                        takeDamage();
                    }
                }
            });

            if (Math.abs(player.x - portal.x) < 30 && Math.abs(player.y - portal.y) < 40) {
                victory();
            }

            player.animFrame += 0.15;
        }

                // Level 3: Volcanic SPLDV Mountain Sky
                const skyGrad = ctx.createLinearGradient(0, 0, 0, h);
                skyGrad.addColorStop(0, '#2a0808');
                skyGrad.addColorStop(0.5, '#7f1d1d');
                skyGrad.addColorStop(1, '#ea580c');
                ctx.fillStyle = skyGrad;
                ctx.fillRect(0, 0, w, h);

                // Volcanic Blood Moon
                ctx.fillStyle = 'rgba(251, 146, 60, 0.85)';
                ctx.beginPath();
                ctx.arc(w - 180 - cameraX * 0.02, 95, 42, 0, Math.PI * 2);
                ctx.fill();

                // Sharp Magma Peaks Background
                ctx.fillStyle = '#450a0a';
                ctx.beginPath();
                ctx.moveTo(0, h);
                for (let x = -50; x <= w + 120; x += 110) {
                    let py = 200 - (x % 220 === 0 ? 110 : 35) - Math.sin(x) * 25;
                    ctx.lineTo(x, py);
                }
                ctx.lineTo(w + 120, h);
                ctx.fill();

                // Jagged Mountain Foreground
                ctx.fillStyle = '#78350f';
                ctx.beginPath();
                ctx.moveTo(0, h);
                for (let x = -30; x <= w + 100; x += 90) {
                    let hy = 320 - Math.sin((x + cameraX * 0.28) * 0.015) * 55;
                    ctx.lineTo(x, hy);
                }
                ctx.lineTo(w + 100, h);
                ctx.fill();

                // Floating Fire Embers
                for (let e = 0; e < 25; e++) {
                    let ex = ((e * 70 + Math.sin(player.animFrame + e) * 20 - cameraX * 0.45) % (w + 100));
                    if (ex < 0) ex += w + 100;
                    let ey = (500 - ((player.animFrame * 22 + e * 35) % 480));
                    ctx.fillStyle = 'rgba(253, 224, 71, 0.9)';
                    ctx.beginPath();
                    ctx.arc(ex, ey, 2.5, 0, Math.PI * 2);
                    ctx.fill();
                }
            }
        }

        function render() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            drawBackground();

            ctx.save();
            ctx.translate(-cameraX, 0);

            platforms.forEach(p => {
                ctx.fillStyle = currentLevel === 1 ? '#22c55e' : (currentLevel === 2 ? '#14b8a6' : '#f97316');
                ctx.fillRect(p.x, p.y, p.w, 6);

                ctx.fillStyle = '#334155';
                ctx.fillRect(p.x, p.y + 6, p.w, p.h - 6);
                ctx.strokeStyle = '#1e293b';
                ctx.strokeRect(p.x, p.y, p.w, p.h);
            });

            coins.forEach(c => {
                if (!c.taken) {
                    ctx.fillStyle = '#f59e0b';
                    ctx.beginPath();
                    ctx.arc(c.x, c.y + Math.sin(player.animFrame) * 3, 8, 0, Math.PI * 2);
                    ctx.fill();
                    ctx.strokeStyle = '#fef08a';
                    ctx.stroke();
                }
            });

            gems.forEach(g => {
                if (!g.taken) {
                    ctx.fillStyle = '#38bdf8';
                    ctx.beginPath();
                    ctx.moveTo(g.x, g.y - 12);
                    ctx.lineTo(g.x + 10, g.y);
                    ctx.lineTo(g.x, g.y + 12);
                    ctx.lineTo(g.x - 10, g.y);
                    ctx.closePath();
                    ctx.fill();
                }
            });

            mathChests.forEach(c => {
                ctx.fillStyle = c.opened ? '#64748b' : '#d97706';
                ctx.fillRect(c.x - 15, c.y - 15, 30, 25);
                ctx.fillStyle = '#fef08a';
                ctx.fillRect(c.x - 4, c.y - 5, 8, 8);
            });

            slimes.forEach(s => {
                if (s.alive) {
                    ctx.fillStyle = '#e11d48';
                    ctx.beginPath();
                    ctx.arc(s.x, s.y, 14, Math.PI, 0);
                    ctx.fill();
                    ctx.fillStyle = '#ffffff';
                    ctx.fillRect(s.x - 6, s.y - 8, 4, 4);
                    ctx.fillRect(s.x + 2, s.y - 8, 4, 4);
                }
            });

            ctx.fillStyle = '#a855f7';
            ctx.beginPath();
            ctx.ellipse(portal.x + 24, portal.y + 32, 20, 32, player.animFrame, 0, Math.PI * 2);
            ctx.fill();

            if (player.invulnerableTimer % 6 < 3) {
                ctx.save();
                ctx.translate(player.x + player.width / 2, player.y + player.height / 2);

                ctx.save();
                ctx.font = 'bold 11px "Plus Jakarta Sans", sans-serif';
                ctx.textAlign = 'center';
                const nameWidth = ctx.measureText(playerName).width;

                ctx.fillStyle = 'rgba(15, 23, 42, 0.85)';
                ctx.beginPath();
                ctx.roundRect(-nameWidth / 2 - 6, -42, nameWidth + 12, 18, 6);
                ctx.fill();
                ctx.strokeStyle = '#38bdf8';
                ctx.lineWidth = 1;
                ctx.stroke();

                ctx.fillStyle = '#fef08a';
                ctx.fillText(playerName, 0, -29);
                ctx.restore();

                drawPlayerSkin(ctx, activeSkin, player.animFrame, player.facing === 'LEFT', hasShield);

                ctx.restore();
            }

            ctx.restore();
        }
